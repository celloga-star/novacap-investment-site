import React, { useState, useEffect } from 'react';

function formatBRL(v: number) {
  return new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL', maximumFractionDigits: 0 }).format(v);
}

export default function Simulator() {
  const [initial, setInitial] = useState<number>(5000);
  const [monthly, setMonthly] = useState<number>(1000);
  const [annual, setAnnual] = useState<number>(12);
  const [years, setYears] = useState<number>(5);
  const [result, setResult] = useState<number>(0);

  useEffect(() => {
    const r = calculate(initial, monthly, annual, years);
    setResult(r);
  }, [initial, monthly, annual, years]);

  function calculate(init: number, aporte: number, anual: number, anos: number) {
    const meses = anos * 12;
    let montante = init;
    const taxaMensal = Math.pow(1 + anual / 100, 1 / 12) - 1;
    for (let i = 0; i < meses; i++) {
      montante = montante * (1 + taxaMensal) + aporte;
    }
    return Math.round(montante);
  }

  return (
    <section id="simulador">
      <div className="grid md:grid-cols-2 gap-6">
        <div className="card">
          <div className="small-muted">Simulador</div>
          <h3 className="text-2xl font-bold mt-1">Planeje seu investimento</h3>

          <div className="mt-4 space-y-4">
            <label className="block">
              <div className="small-muted">Valor inicial</div>
              <input type="number" value={initial} onChange={(e) => setInitial(Number(e.target.value))} className="mt-1 w-full rounded-lg p-3 bg-transparent border border-white/6" />
            </label>

            <label className="block">
              <div className="small-muted">Aporte mensal</div>
              <input type="number" value={monthly} onChange={(e) => setMonthly(Number(e.target.value))} className="mt-1 w-full rounded-lg p-3 bg-transparent border border-white/6" />
            </label>

            <label className="grid grid-cols-2 gap-3">
              <div>
                <div className="small-muted">Rentabilidade anual (%)</div>
                <input type="number" value={annual} onChange={(e) => setAnnual(Number(e.target.value))} className="mt-1 w-full rounded-lg p-3 bg-transparent border border-white/6" />
              </div>
              <div>
                <div className="small-muted">Horizonte (anos)</div>
                <input type="number" value={years} onChange={(e) => setYears(Number(e.target.value))} className="mt-1 w-full rounded-lg p-3 bg-transparent border border-white/6" />
              </div>
            </label>
          </div>
        </div>

        <div className="card">
          <div className="small-muted">Resultado</div>
          <h3 className="text-2xl font-extrabold mt-1">{formatBRL(result)}</h3>
          <p className="small-muted mt-2">Valor estimado em {years} anos</p>

          <div className="mt-6">
            <button className="px-4 py-3 rounded-lg bg-gradient-to-br from-primary to-[#82f1d3] text-slate-900 font-bold">Simular estratégia</button>
            <button className="ml-3 px-4 py-3 rounded-lg border border-white/6 text-slate-200">Salvar simulador</button>
          </div>

          <div className="mt-6 small-muted text-sm">
            Atenção: este simulador é meramente ilustrativo e não representa retorno garantido.
          </div>
        </div>
      </div>
    </section>
  );
}
